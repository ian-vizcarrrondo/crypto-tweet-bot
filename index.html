const EMOJIS = { 'bitcoin':'₿','ethereum':'Ξ','solana':'◎','binancecoin':'🔶','ripple':'💧', 'dogecoin':'🐶','avalanche-2':'🔺','chainlink':'🔗','polkadot':'⚫','matic-network':'🟣', 'shiba-inu':'🐕','pepe':'🐸','floki':'⚡','dogwifcoin':'🎩','bonk':'🔨' }; let sparklineCharts = {}; let countdown = 60; let countdownInterval = null; function formatPrice(p) { if (p >= 1000) return '$' + p.toLocaleString('en-US', {minimumFractionDigits:2, maximumFractionDigits:2}); if (p >= 1) return '$' + p.toFixed(2); if (p >= 0.01) return '$' + p.toFixed(4); if (p >= 0.0001) return '$' + p.toFixed(6); return '$' + p.toFixed(8); } function formatLarge(v) { if (!v) return '—'; if (v >= 1e12) return '$' + (v/1e12).toFixed(2) + 'T'; if (v >= 1e9) return '$' + (v/1e9).toFixed(1) + 'B'; if (v >= 1e6) return '$' + (v/1e6).toFixed(1) + 'M'; return '$' + v.toLocaleString(); } function cc(v) { if (v==null) return 'neutral'; return v>=0?'green':'red'; } function ct(v) { if (v==null) return '—'; return (v>=0?'+':'')+v.toFixed(2)+'%'; } function fngColor(v) { if (v<=25) return '#ff4757'; if (v<=45) return '#ff8c42'; if (v<=55) return '#f5a623'; if (v<=75) return '#7bc67e'; return '#00d4aa'; } function drawSparkline(id, prices, positive) { if (sparklineCharts[id]) { sparklineCharts[id].destroy(); delete sparklineCharts[id]; } const canvas = document.getElementById(id); if (!canvas || !prices || prices.length === 0) return; const color = positive ? '#00d4aa' : '#ff4757'; sparklineCharts[id] = new Chart(canvas, { type: 'line', data: { labels: prices.map((_,i)=>i), datasets: [{ data: prices, borderColor: color, borderWidth: 1.5, pointRadius: 0, fill: true, backgroundColor: color+'18', tension: 0.4 }] }, options: { responsive:false, animation:false, plugins:{legend:{display:false},tooltip:{enabled:false}}, scales:{x:{display:false},y:{display:false}} } }); } function renderTable(bodyId, coins) { const tbody = document.getElementById(bodyId); tbody.innerHTML = ''; coins.forEach((coin, idx) => { const sid = `spark-${coin.id}-${bodyId}`; const row = document.createElement('tr'); row.innerHTML = ` ${idx+1}
${coin.emoji || EMOJIS[coin.id] || '🪙'}
${coin.name}
${coin.symbol}
${formatPrice(coin.price)} ${ct(coin.change_1h)} ${ct(coin.change_24h)} ${ct(coin.change_7d)} ${formatLarge(coin.market_cap)} `; tbody.appendChild(row); if (coin.sparkline && coin.sparkline.length > 0) { setTimeout(() => drawSparkline(sid, coin.sparkline, (coin.change_7d||0) >= 0), 0); } }); } async function fetchAndRender() { try { const res = await fetch('./prices.json?t=' + Date.now()); if (!res.ok) throw new Error('prices.json not found yet — run the workflow first'); const data = await res.json(); // Stats bar const totalMktCap = [...data.main_coins, ...data.meme_coins.filter(c=>c.id!=='dogecoin')] .reduce((s,c) => s + (c.market_cap||0), 0); const btcMktCap = data.main_coins.find(c=>c.id==='bitcoin')?.market_cap || 0; const btcDom = totalMktCap > 0 ? (btcMktCap/totalMktCap*100).toFixed(1) : '—'; const totalVol = [...data.main_coins].reduce((s,c) => s + (c.volume_24h||0), 0); let fngHtml = ''; if (data.fear_greed) { const fc = fngColor(parseInt(data.fear_greed.value)); fngHtml = `
${data.fear_greed.value}
Fear & Greed
${data.fear_greed.value_classification}
`; } document.getElementById('stats-bar').innerHTML = `
Market Cap
${formatLarge(totalMktCap)}
BTC Dominance
${btcDom}%
24h Volume
${formatLarge(totalVol)}
${fngHtml} `; // Highlights const all = [...data.main_coins, ...data.meme_coins.filter(c=>c.id!=='dogecoin')]; const sorted = [...all].sort((a,b) => (b.change_24h||0) - (a.change_24h||0)); const top = sorted[0], bot = sorted[sorted.length-1]; document.getElementById('highlights').style.display = 'flex'; document.getElementById('highlights').innerHTML = `
🚀
Top Gainer 24h
${top.emoji} ${top.name}
${ct(top.change_24h)}
💥
Biggest Drop 24h
${bot.emoji} ${bot.name}
${ct(bot.change_24h)}
`; document.getElementById('main-section').style.display = 'block'; document.getElementById('meme-section').style.display = 'block'; renderTable('main-coins-body', data.main_coins); renderTable('meme-coins-body', data.meme_coins); const updated = new Date(data.updated); document.getElementById('last-updated').textContent = 'Last updated: ' + updated.toLocaleTimeString(); startCountdown(); } catch(err) { console.error(err); document.getElementById('stats-bar').innerHTML = `
⚠️ ${err.message}
`; startCountdown(); } } function startCountdown() { if (countdownInterval) clearInterval(countdownInterval); countdown = 60; document.getElementById('countdown').textContent = '60s'; countdownInterval = setInterval(() => { countdown--; document.getElementById('countdown').textContent = countdown + 's'; if (countdown <= 0) { clearInterval(countdownInterval); fetchAndRender(); } }, 1000); } fetchAndRender();
